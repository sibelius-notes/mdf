{{- /* Study time estimation — handles CJK and Latin prose separately: • CJK characters → 300 cpm (academic Chinese reading speed) • Latin/other words → 180 wpm (English reading speed) Then adds overhead for inline/display math and LaTeX command density. Hugo's countwords uses whitespace-splitting, which massively undercounts Chinese text (no spaces between chars), so we count CJK chars via findRE. */ -}} {{- $raw := .RawContent -}} {{- $plain := .Plain -}} {{- $inlinePattern := `(?s)\\\(.*?\\\)` -}} {{- $displayPattern := `(?s)\\\[.*?\\\]` -}} {{- $envPattern := `(?s)\\begin\{[A-Za-z*]+\}.*?\\end\{[A-Za-z*]+\}` -}} {{- $inlineMatches := findRE $inlinePattern $raw -}} {{- $displayMatches := findRE $displayPattern $raw -}} {{- $rawWithoutDelimitedMath := replaceRE $displayPattern " " (replaceRE $inlinePattern " " $raw) -}} {{- $envMatches := findRE $envPattern $rawWithoutDelimitedMath -}} {{- $plainWithoutInlineMath := replaceRE $inlinePattern " " $plain -}} {{- $plainWithoutDisplayMath := replaceRE $displayPattern " " $plainWithoutInlineMath -}} {{- $prosePlain := replaceRE $envPattern " " $plainWithoutDisplayMath -}} {{- $mathContent := delimit (append (append $inlineMatches $displayMatches) $envMatches) " " -}} {{- /* Count CJK characters (CJK Unified Ideographs blocks) */ -}} {{- $cjkChars := len (findRE `[\x{4e00}-\x{9fff}\x{3400}-\x{4dbf}\x{f900}-\x{faff}]` $prosePlain) -}} {{- /* Strip CJK from prosePlain so countwords only sees Latin/numeric tokens */ -}} {{- $latinPlain := replaceRE `[\x{4e00}-\x{9fff}\x{3400}-\x{4dbf}\x{f900}-\x{faff}]` "" $prosePlain -}} {{- $latinWords := countwords $latinPlain -}} {{- $inlineEquations := len $inlineMatches -}} {{- $displayEquations := add (len $displayMatches) (len $envMatches) -}} {{- $mathCommands := len (findRE `\\[A-Za-z]+` $mathContent) -}} {{- /* 300 cpm for CJK; 180 wpm for Latin */ -}} {{- $cjkSeconds := div (mul $cjkChars 60) 300 -}} {{- $latinSeconds := div (mul $latinWords 60) 180 -}} {{- $proseSeconds := add $cjkSeconds $latinSeconds -}} {{- $inlineSeconds := mul $inlineEquations 2 -}} {{- $displaySeconds := mul $displayEquations 20 -}} {{- $commandSeconds := div $mathCommands 2 -}} {{- $studySeconds := add (add $proseSeconds $inlineSeconds) (add $displaySeconds $commandSeconds) -}} {{- $studyMinutes := div (add $studySeconds 59) 60 -}} {{- /* proseWords kept for display: CJK chars + latin words */ -}} {{- $proseWords := add $cjkChars $latinWords -}} {{- return (dict "display_equations" $displayEquations "inline_equations" $inlineEquations "math_commands" $mathCommands "minutes" $studyMinutes "prose_words" $proseWords ) -}}
